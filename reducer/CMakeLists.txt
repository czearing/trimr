cmake_minimum_required(VERSION 3.16)
project(trimr_reduce CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(nifly GIT_REPOSITORY https://github.com/ousnius/nifly.git GIT_TAG cca0a770094bb962fb28ea1fec5ea903e68fda8e)
FetchContent_Declare(meshoptimizer GIT_REPOSITORY https://github.com/zeux/meshoptimizer.git GIT_TAG b2330430776a95a1545f03fc7bf16927f8cadc59)
FetchContent_GetProperties(nifly)
if(NOT nifly_POPULATED)
  FetchContent_Populate(nifly)
  add_subdirectory(${nifly_SOURCE_DIR} ${nifly_BINARY_DIR} EXCLUDE_FROM_ALL)
endif()
FetchContent_GetProperties(meshoptimizer)
if(NOT meshoptimizer_POPULATED)
  FetchContent_Populate(meshoptimizer)
  file(GLOB MESHOPT_SRC ${meshoptimizer_SOURCE_DIR}/src/*.cpp)
  add_library(meshopt STATIC ${MESHOPT_SRC})
  target_include_directories(meshopt PUBLIC ${meshoptimizer_SOURCE_DIR}/src)
endif()
add_executable(trimr-reduce main.cpp)
target_include_directories(trimr-reduce PRIVATE ${nifly_SOURCE_DIR}/include ${nifly_SOURCE_DIR}/external)
target_link_libraries(trimr-reduce PRIVATE nifly meshopt)
